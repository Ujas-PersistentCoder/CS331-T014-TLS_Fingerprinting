cmake_minimum_required(VERSION 3.22)
project(ech_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BORINGSSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.local/boringssl" CACHE PATH "Pinned BoringSSL checkout")
set(BORINGSSL_BUILD "${CMAKE_CURRENT_SOURCE_DIR}/.local/boringssl-build" CACHE PATH "Separate BoringSSL build")
add_executable(ech_lab lab.cpp)
target_include_directories(ech_lab PRIVATE "${BORINGSSL_ROOT}/include")
target_link_libraries(ech_lab PRIVATE "${BORINGSSL_BUILD}/libssl.a" "${BORINGSSL_BUILD}/libcrypto.a")
target_compile_options(ech_lab PRIVATE -Wall -Wextra -Wpedantic)
